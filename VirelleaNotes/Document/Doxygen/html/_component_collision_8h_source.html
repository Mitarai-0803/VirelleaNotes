265 bool CalcPush(ComponentCollisionPtr collision, float3 vec, float3* push, float3* other_push) const
266 {
267 float mass1 = GetMass();
268 float mass2 = collision->GetMass();
269 if(mass1 < 0 && mass2 < 0) {
270 // 両方不動のものがぶつかった
271 *push = vec * 0.5f;
272 *other_push = -vec * 0.5f;
273 return false;
274 }
275 else if(mass1 < 0) {
276 // 相手に跳ね返す
277 *push = {0, 0, 0};
278 *other_push = -vec;
279 }
280 else if(mass2 < 0) {
281 // 相手に跳ね返す
282 *push = vec;
283 *other_push = {0, 0, 0};
284 }
285 else {
286 // 重さに合わせて移動させる
287 mass1 += 0.001f;
288 mass2 += 0.001f;
289 float all = mass1 + mass2;
290 *push = vec * mass2 / all;
291 *other_push = -vec * mass1 / all;
292 }
293 return true;
294 }

296 bool IsGroupHit(ComponentCollisionPtr collision) const
297 {
298 // コリジョンが当たらない設定になっている
299 if(collision_status_.is(CollisionBit::DisableHit) || collision->collision_status_.is(CollisionBit::DisableHit))
300 return false;

302 // 自分か相手のコリジョンタイプが不定
303 if(collision_type_ == CollisionType::NONE || collision->collision_type_ == CollisionType::NONE)
304 return false;

307 if(((u32)collision_group_ & collision->collision_hit_) == 0)

311 if(((u32)collision->collision_group_ & collision_hit_) == 0)
312 return false;

314 // 当たる設定となる
315 return true;
